Entrenamiento distribuido con FSDP2, DCP y torchtitan

El stack PyTorch-nativo detrás de torchtitan, OLMo-core y el backend FSDP2 de accelerate: un script FSDP2 que se entiende línea por línea, checkpoints DCP que se reanudan con otro número de GPUs, multi-nodo en SLURM con requeue automático, la misma receta en accelerate y una receta torchtitan para continual pretraining de Llama 3.1 8B en español y portugués. Complementa la guía de accelerate + DeepSpeed.

1. Mapa mental: qué paralelismo necesitas

FSDP2 (torch.distributed.fsdp.fully_shard) es la API que PyTorch recomienda hoy para código nuevo en lugar de FSDP1 (FullyShardedDataParallel). En vez de aplanar cada módulo en un FlatParameter, cada parámetro pasa a ser un DTensor shardeado en la dimensión 0. Eso permite congelar pesos (LoRA), mezclar FP8 y BF16 en el mismo modelo y guardar checkpoints sin comunicación extra. Es el equivalente a ZeRO-3 de DeepSpeed, pero nativo, y se compone con tensor parallel (TP), pipeline parallel (PP) y context parallel (CP) sobre un mismo DeviceMesh.

2. FSDP2 desde cero: continual pretraining es/pt en ~170 líneas

El script hace continual pretraining de Qwen/Qwen3-0.6B-Base sobre una mezcla de FineWeb-2 (spa_Latn y por_Latn) con 15% de replay en inglés de FineWeb-Edu para no olvidar capacidades. Lo esencial: fully_shard primero en cada bloque transformer y después en la raíz, MixedPrecisionPolicy con cómputo en BF16 y reducción de gradientes en FP32, y checkpoints DCP con el estado del modelo, del optimizador, del scheduler y del stream de datos.

python · fsdp2_min.py
import os
import time
from pathlib import Path

import torch
import torch.distributed as dist
import torch.distributed.checkpoint as dcp
from datasets import interleave_datasets, load_dataset
from datasets.distributed import split_dataset_by_node
from torch.distributed.checkpoint.state_dict import (
    StateDictOptions, get_model_state_dict, get_state_dict, set_state_dict,
)
from torch.distributed.checkpoint.stateful import Stateful
from torch.distributed.device_mesh import init_device_mesh
from torch.distributed.fsdp import MixedPrecisionPolicy, fully_shard
from torch.distributed.tensor import DTensor
from transformers import AutoModelForCausalLM, AutoTokenizer

MODEL_ID = os.environ.get("MODEL_ID", "Qwen/Qwen3-0.6B-Base")
SEQ_LEN, MICRO_BS, GRAD_ACCUM = 4096, 4, 4      # 8 GPUs → 128 secuencias = 0.5M tokens por paso
TOTAL_STEPS, WARMUP, DECAY = 5_000, 200, 1_000  # schedule WSD: warmup, estable, decay final
PEAK_LR = 3e-5                                  # LR bajo: es continual pretraining, no desde cero
SAVE_EVERY = 250
PEAK_FLOPS = 989e12                             # BF16 denso por GPU: H100 SXM 989e12, A100 312e12
CKPT_DIR = Path(os.environ.get("CKPT_DIR", "checkpoints/qwen3-0.6b-es-pt"))
STOP_FLAG = Path(os.environ.get("STOP_FLAG", "STOP_AND_REQUEUE"))


class PackedStream:
    """Tokeniza documentos y los concatena (EOS entre docs) en bloques de seq_len tokens."""

    def __init__(self, ds, tok, seq_len):
        self.ds, self.tok, self.seq_len = ds, tok, seq_len
        self.buffer: list[int] = []

    def __iter__(self):
        for ex in self.ds:
            self.buffer += self.tok(ex["text"], add_special_tokens=False)["input_ids"]
            self.buffer.append(self.tok.eos_token_id)
            while len(self.buffer) >= self.seq_len:
                chunk, self.buffer = self.buffer[: self.seq_len], self.buffer[self.seq_len :]
                yield torch.tensor(chunk, dtype=torch.long)

    def state_dict(self):
        return {"ds": self.ds.state_dict(), "buffer": list(self.buffer)}

    def load_state_dict(self, sd):
        self.ds.load_state_dict(sd["ds"])
        self.buffer = sd["buffer"]


def build_stream(tok, rank: int, world: int) -> PackedStream:
    def src(repo, name, cfg_split="train"):
        ds = load_dataset(repo, name=name, split=cfg_split, streaming=True)
        return ds.select_columns(["text"])  # interleave exige las mismas columnas

    es = src("HuggingFaceFW/fineweb-2", "spa_Latn")
    pt = src("HuggingFaceFW/fineweb-2", "por_Latn")
    en = src("HuggingFaceFW/fineweb-edu", "sample-10BT")       # replay en inglés
    mix = interleave_datasets([es, pt, en], probabilities=[0.5, 0.35, 0.15], seed=42)
    mix = split_dataset_by_node(mix, rank=rank, world_size=world)
    return PackedStream(mix, tok, SEQ_LEN)


def batches(stream, bs):
    it = iter(stream)
    while True:
        yield torch.stack([next(it) for _ in range(bs)])


class AppState(Stateful):
    """DCP llama a state_dict()/load_state_dict(): cada rank guarda y lee solo sus shards."""

    def __init__(self, model, optim, sched):
        self.model, self.optim, self.sched = model, optim, sched

    def state_dict(self):
        msd, osd = get_state_dict(self.model, self.optim)
        return {"model": msd, "optim": osd, "sched": self.sched.state_dict()}

    def load_state_dict(self, sd):
        set_state_dict(self.model, self.optim, model_state_dict=sd["model"], optim_state_dict=sd["optim"])
        self.sched.load_state_dict(sd["sched"])


def latest_checkpoint(root: Path):
    done = sorted(p for p in root.glob("step_*") if (p / ".metadata").exists())  # .metadata = guardado completo
    return done[-1] if done else None


def wsd(step: int) -> float:
    if step < WARMUP:
        return (step + 1) / WARMUP
    if step < TOTAL_STEPS - DECAY:
        return 1.0
    return max(0.0, (TOTAL_STEPS - step) / DECAY)


def export_hf(model, tok, rank: int, out_dir: str = "export-hf"):
    full = get_model_state_dict(model, options=StateDictOptions(full_state_dict=True, cpu_offload=True))
    if rank == 0:
        hf = AutoModelForCausalLM.from_pretrained(MODEL_ID, dtype=torch.bfloat16)
        missing, unexpected = hf.load_state_dict({k: v.to(torch.bfloat16) for k, v in full.items()}, strict=False)
        assert not unexpected, unexpected
        print("faltantes (deben ser solo pesos atados):", missing)
        hf.save_pretrained(out_dir)       # safetensors por defecto
        tok.save_pretrained(out_dir)
    dist.barrier()


def main():
    torch.cuda.set_device(int(os.environ["LOCAL_RANK"]))
    dist.init_process_group(backend="cpu:gloo,cuda:nccl")  # gloo para el guardado asíncrono en CPU
    rank, world = dist.get_rank(), dist.get_world_size()
    # 1D = FSDP puro. HSDP multi-nodo: init_device_mesh("cuda", (n_nodos, 8), mesh_dim_names=("replicate", "shard"))
    mesh = init_device_mesh("cuda", (world,), mesh_dim_names=("shard",))

    tok = AutoTokenizer.from_pretrained(MODEL_ID)
    # Modelo chico: cada rank lo carga en CPU en FP32 (pesos maestros); fully_shard lo mueve a la GPU ya shardeado.
    model = AutoModelForCausalLM.from_pretrained(MODEL_ID, dtype=torch.float32)
    model.config.use_cache = False
    model.gradient_checkpointing_enable(gradient_checkpointing_kwargs={"use_reentrant": False})
    n_params = sum(p.numel() for p in model.parameters())

    mp = MixedPrecisionPolicy(param_dtype=torch.bfloat16, reduce_dtype=torch.float32)
    for layer in model.model.layers:
        fully_shard(layer, mesh=mesh, mp_policy=mp)   # una unidad de all-gather por bloque
    fully_shard(model, mesh=mesh, mp_policy=mp)       # raíz: embeddings, norm final y lm_head
    model.train()

    optim = torch.optim.AdamW(model.parameters(), lr=PEAK_LR, betas=(0.9, 0.95), weight_decay=0.1)
    sched = torch.optim.lr_scheduler.LambdaLR(optim, wsd)
    stream = build_stream(tok, rank, world)
    app = AppState(model, optim, sched)

    last = latest_checkpoint(CKPT_DIR)
    if last is not None:
        dcp.load({"app": app}, checkpoint_id=str(last))  # resharding automático si cambió el nº de GPUs
        data_file = last / f"data_rank{rank}.pt"
        if data_file.exists():
            stream.load_state_dict(torch.load(data_file, weights_only=False))
        if rank == 0:
            print(f"Reanudando desde {last}")

    step, pending, t0 = sched.last_epoch, None, time.perf_counter()
    data = batches(stream, MICRO_BS)
    while step < TOTAL_STEPS:
        loss_acc = torch.zeros((), device="cuda")
        for micro in range(GRAD_ACCUM):
            model.set_requires_gradient_sync(micro == GRAD_ACCUM - 1)  # reduce-scatter solo al final
            ids = next(data).cuda(non_blocking=True)
            loss = model(input_ids=ids, labels=ids).loss / GRAD_ACCUM
            loss.backward()
            loss_acc += loss.detach()
        gnorm = torch.nn.utils.clip_grad_norm_(model.parameters(), 1.0)  # norma global aun con DTensor
        if isinstance(gnorm, DTensor):
            gnorm = gnorm.full_tensor()
        optim.step()
        sched.step()
        optim.zero_grad(set_to_none=True)
        step += 1

        if step % 10 == 0:
            dist.all_reduce(loss_acc, op=dist.ReduceOp.AVG)
            dt, t0 = time.perf_counter() - t0, time.perf_counter()
            tok_s = 10 * GRAD_ACCUM * MICRO_BS * SEQ_LEN / dt          # tokens/s por GPU
            mfu = tok_s * 6 * n_params / PEAK_FLOPS                      # ignora FLOPs de atención y recompute
            if rank == 0:
                print(f"step {step} | loss {loss_acc.item():.4f} | gnorm {gnorm.item():.2f} | "
                      f"lr {sched.get_last_lr()[0]:.2e} | {tok_s:,.0f} tok/s/GPU | MFU {mfu:.1%}")

        stop = torch.tensor(float(STOP_FLAG.exists()), device="cuda")
        dist.all_reduce(stop, op=dist.ReduceOp.MAX)                     # todos los ranks deciden lo mismo
        if step % SAVE_EVERY == 0 or stop.item() > 0:
            if pending is not None:
                pending.result()                                         # un guardado a la vez
            path = CKPT_DIR / f"step_{step:07d}"
            path.mkdir(parents=True, exist_ok=True)
            torch.save(stream.state_dict(), path / f"data_rank{rank}.pt")
            if stop.item() > 0:
                dcp.save({"app": app}, checkpoint_id=str(path))          # síncrono: SLURM está por cortar
                break
            pending = dcp.async_save({"app": app}, checkpoint_id=str(path))

    if pending is not None:
        pending.result()
    if step >= TOTAL_STEPS:
        export_hf(model, tok, rank)
    dist.destroy_process_group()


if __name__ == "__main__":
    main()

3. Memoria y eficiencia: AC, offload y MFU

Con FSDP2 la memoria por GPU es aproximadamente (pesos + gradientes + Adam) / N GPUs + activaciones del bloque actual. Las palancas, en orden de costo:

PalancaAhorroCosto
Activation checkpointing (use_reentrant=False)Activaciones de ~N capas a ~1~30% más cómputo (un forward extra)
reshard_after_forward=False en la raíz o en el último bloqueEvita un all-gather en el backwardMás memoria: mantiene ese bloque completo
CPUOffloadPolicy() en fully_shard(..., offload_policy=...)Parámetros y estados de Adam a CPUMucho más lento; último recurso para GPUs de 24 GB
Liger Kernel o Cut Cross-EntropyEl pico de logits (vocab × seq) de la capa de salidaUna dependencia más; clave si extendiste el vocabulario es/pt

El MFU (model FLOPs utilization) se calcula como tokens/s × 6N / FLOP/s pico, con N parámetros. En el log del script lo calculamos por GPU. Como referencia, Llama 3 405B reportó 38-43% en BF16 sobre H100. Un modelo de 0.6B va a dar bastante menos, porque la comunicación pesa más en proporción al cómputo, y eso es normal. Para ver dónde se va la memoria:

4. Checkpoints con DCP: asíncronos, reshardeables y exportables

PyTorch Distributed Checkpoint (DCP) guarda cada shard en paralelo desde su rank, junto con un archivo .metadata que describe la partición. Hay tres propiedades que lo hacen la base de corridas largas:

Asíncrono: dcp.async_save copia el estado a CPU y escribe a disco en otro hilo mientras el entrenamiento sigue. Por eso el script espera pending.result() antes del siguiente guardado. Resharding: al cargar, cada rank pide los trozos que corresponden a su sharding actual, así que un checkpoint de 16 GPUs se reanuda en 8 sin convertir nada. Atomicidad práctica: .metadata se escribe al final, y por eso latest_checkpoint ignora los directorios incompletos que deja un job que murió a mitad del guardado.

Estado del dataloader al cambiar de topología: el modelo y el optimizador se reshardean solos, pero el stream de datos se guarda por rank (data_rank{i}.pt). Si cambias el número de GPUs, esos archivos ya no corresponden y el script reinicia el stream desde el principio, así que vas a repetir datos. En producción conviene saltar por número de documentos consumidos o cambiar la semilla de la mezcla y registrarlo.

Para llevar un checkpoint intermedio a Hugging Face sin levantar el cluster, se convierte DCP a un state_dict normal en una sola máquina:

python · dcp_to_hf.py
import torch
from torch.distributed.checkpoint.format_utils import dcp_to_torch_save
from transformers import AutoModelForCausalLM, AutoTokenizer

BASE = "Qwen/Qwen3-0.6B-Base"
CKPT = "checkpoints/qwen3-0.6b-es-pt/step_0002500"

dcp_to_torch_save(CKPT, "full_step2500.pt")              # consolida todos los shards (necesita RAM ~ tamaño total)
state = torch.load("full_step2500.pt", weights_only=False)
model_sd = state["app"]["model"]                          # misma estructura que AppState.state_dict()

model = AutoModelForCausalLM.from_pretrained(BASE, dtype=torch.bfloat16)
missing, unexpected = model.load_state_dict({k: v.to(torch.bfloat16) for k, v in model_sd.items()}, strict=False)
print("missing:", missing, "| unexpected:", unexpected)
model.save_pretrained("export-step2500")

5. Multi-nodo en SLURM con guardado antes del timeout y requeue

Los clusters académicos de la región suelen ser compartidos y limitan los jobs a 24-48 h. La receta: --signal=B:USR1@300 le avisa al script batch 5 minutos antes del límite. El trap crea un archivo bandera que el loop detecta (todos los ranks se ponen de acuerdo con un all_reduce), se hace un checkpoint síncrono, el proceso sale y el batch se re-encola solo con scontrol requeue. Al volver a arrancar, latest_checkpoint encuentra el último guardado. El lanzamiento sigue el patrón de multinode_trainer.slurm de torchtitan: un torchrun por nodo con rendezvous c10d.

bash · train_multinode.sbatch
#!/bin/bash
#SBATCH --job-name=cpt-es-pt
#SBATCH --nodes=2
#SBATCH --ntasks-per-node=1
#SBATCH --gpus-per-node=8
#SBATCH --cpus-per-task=64
#SBATCH --time=24:00:00
#SBATCH --signal=B:USR1@300
#SBATCH --requeue
#SBATCH --output=logs/%x-%j.out

# Sin "set -e": "wait" devuelve >128 cuando lo interrumpe la señal y no queremos abortar ahí
set -uo pipefail

nodes=( $(scontrol show hostnames "$SLURM_JOB_NODELIST") )
HEAD_IP=$(srun --nodes=1 --ntasks=1 -w "${nodes[0]}" hostname --ip-address | awk '{print $1}')

# Ajusta a tu red: revisa interfaces con "ip a" y HCAs con "ibv_devices"
export NCCL_SOCKET_IFNAME=ib0
export NCCL_IB_HCA=mlx5
export PYTORCH_ALLOC_CONF=expandable_segments:True
export TORCH_NCCL_TRACE_BUFFER_SIZE=2000
export TORCH_NCCL_DUMP_ON_TIMEOUT=true
export CKPT_DIR="/scratch/$USER/ckpt/qwen3-0.6b-es-pt"
export STOP_FLAG="$SLURM_SUBMIT_DIR/STOP_$SLURM_JOB_ID"
rm -f "$STOP_FLAG"

trap 'echo "[$(date)] USR1: checkpoint y requeue"; touch "$STOP_FLAG"' USR1

srun --kill-on-bad-exit=1 torchrun \
  --nnodes "$SLURM_NNODES" --nproc_per_node 8 \
  --rdzv_id "$SLURM_JOB_ID" --rdzv_backend c10d --rdzv_endpoint "$HEAD_IP:29500" \
  fsdp2_min.py &
SRUN_PID=$!

# bash solo ejecuta el trap entre comandos: srun va en background y esperamos en un loop
wait "$SRUN_PID"
while kill -0 "$SRUN_PID" 2>/dev/null; do
  wait "$SRUN_PID"
done

if [ -f "$STOP_FLAG" ]; then
  rm -f "$STOP_FLAG"
  scontrol requeue "$SLURM_JOB_ID"
fi
HSDP entre nodos: con 2 nodos y red inter-nodo más lenta que NVLink, cambia el mesh a init_device_mesh("cuda", (2, 8), mesh_dim_names=("replicate", "shard")). fully_shard con un mesh 2D shardea dentro del nodo y replica entre nodos, así que entre nodos solo viaja un all-reduce de gradientes. Para 8B en 2 nodos suele bastar. Para 70B el shard dentro del nodo ya no alcanza y hay que sumar TP.
Ajusta los 300 segundos: el checkpoint síncrono de un 0.6B tarda segundos, pero uno de 8B con optimizador (~96 GB) puede tardar minutos en un filesystem lento. Mide cuánto tarda dcp.save en tu cluster y deja al menos el doble de margen en --signal.

6. La misma receta con accelerate (FSDP2 + ND-parallel)

Si prefieres quedarte en el ecosistema Hugging Face, accelerate expone FSDP2 con fsdp_version: 2. fsdp_reshard_after_forward reemplaza a fsdp_sharding_strategy; se eliminaron fsdp_backward_prefetch, fsdp_sync_module_states y fsdp_use_orig_params, y SHARDED_STATE_DICT (DCP por debajo) es el default. Para convertir un YAML viejo existe accelerate to-fsdp2. El bloque parallelism_config combina réplica, shard, TP y CP.

python · train_accelerate.py (importa PackedStream y build_stream de fsdp2_min.py)
import torch
from accelerate import Accelerator
from transformers import AutoModelForCausalLM, AutoTokenizer

from fsdp2_min import MODEL_ID, MICRO_BS, batches, build_stream, wsd

acc = Accelerator()                          # lee FSDP2 y parallelism_config del YAML
tok = AutoTokenizer.from_pretrained(MODEL_ID)
model = AutoModelForCausalLM.from_pretrained(MODEL_ID, dtype=torch.float32)
model.config.use_cache = False
optim = torch.optim.AdamW(model.parameters(), lr=3e-5, betas=(0.9, 0.95), weight_decay=0.1)
model, optim = acc.prepare(model, optim)     # con FSDP2, modelo y optimizador se preparan juntos
sched = torch.optim.lr_scheduler.LambdaLR(optim, wsd)

stream = build_stream(tok, acc.process_index, acc.num_processes)
model.train()
for step, ids in enumerate(batches(stream, MICRO_BS), start=1):
    ids = ids.to(acc.device)
    loss = model(input_ids=ids, labels=ids).loss
    acc.backward(loss)
    acc.clip_grad_norm_(model.parameters(), 1.0)
    optim.step()
    sched.step()
    optim.zero_grad()
    if step % 10 == 0:
        acc.print(f"step {step} | loss {loss.item():.4f}")
    if step % 250 == 0:
        acc.save_state(f"ckpt-accelerate/step_{step:07d}")   # sharded (DCP); reanudar con acc.load_state(...)
    if step == 5_000:
        break

7. torchtitan v0.3: receta de continual pretraining Llama 3.1 8B es/pt

torchtitan (paper en ICLR 2025) es la plataforma PyTorch-nativa de referencia para pretraining. Compone FSDP2, TP (incluido async TP), PP y CP sobre un DeviceMesh, con Float8/MXFP8, torch.compile, activation checkpointing selectivo y DCP asíncrono. Desde v0.3 las configs TOML desaparecieron. Ahora una config es una función Python que devuelve Trainer.Config y se elige con MODULE/CONFIG. Los datos van sobre Grain (HuggingFaceStreamingSource, DatasetMixConfig, ConcatThenSplitPackingConfig).

from torchtitan.components.checkpointer import CheckpointManager
from torchtitan.components.data import ConcatThenSplitPackingConfig, GrainDataLoader
from torchtitan.components.data.dataset import DatasetMixConfig, SingleDatasetConfig, WeightedDataset
from torchtitan.components.data.sources import HuggingFaceStreamingSource
from torchtitan.components.optim import AdamW, Optim, OptimizersContainer
from torchtitan.config.parallelism import ParallelismConfig
from torchtitan.hf_datasets.text_datasets import TextProcessor
from torchtitan.trainer import Trainer
from torchtitan_recipes.models.llama3 import llama3_8b


def _web(path: str, name: str) -> SingleDatasetConfig:
    return SingleDatasetConfig(
        source=HuggingFaceStreamingSource.Config(path=path, name=name, split="train"),
        processor=TextProcessor.Config(),
        post_filters=(lambda sample: sample is not None,),
    )


def llama3_8b_latam_cpt(seq_len: int = 8192) -> Trainer.Config:
    config = llama3_8b(seq_len=seq_len)                    # parte de la receta oficial
    config.hf_assets_path = "./assets/hf/Llama-3.1-8B"
    config.dataloader = GrainDataLoader.Config(
        dataset=ConcatThenSplitPackingConfig(
            dataset=DatasetMixConfig(datasets=(
                WeightedDataset(dataset=_web("HuggingFaceFW/fineweb-2", "spa_Latn"), weight=0.50),
                WeightedDataset(dataset=_web("HuggingFaceFW/fineweb-2", "por_Latn"), weight=0.35),
                WeightedDataset(dataset=_web("HuggingFaceFW/fineweb-edu", "sample-10BT"), weight=0.15),
            )),
        ),
    )
    config.parallelism = ParallelismConfig(data_parallel_shard_degree=8, tensor_parallel_degree=1)
    config.optim = Optim.Config(
        optimizer=OptimizersContainer.Config(optimizers=[AdamW.Config(pattern=r".*", lr=3e-5)]),
    )
    config.training.steps = 20_000
    config.checkpointer = CheckpointManager.Config(
        folder="checkpoint-latam",
        interval=500,
        initial_load_in_hf=True,       # arranca desde los safetensors de HF
        last_save_in_hf=True,          # el último guardado sale en formato HF, listo para vLLM
        last_save_model_only=True,
        export_dtype="bfloat16",
        async_mode="async",            # DCP asíncrono ("async_with_pinned_mem" es más rápido)
    )
    # Float8/MXFP8, torch.compile y warmup/decay del LR: revisa los nombres exactos de esos
    # campos con --print-config en tu versión (la API de configs cambia rápido en main).
    return config

8. Contexto largo: context parallelism

Para extender el contexto de 8K a 32K-128K (por ejemplo, para documentos legales o actas en español) las activaciones de atención dejan de caber aunque el modelo quepa. Context parallelism reparte la secuencia entre GPUs:

VarianteCómo comunicaLímiteDónde está
Ring AttentionCircula bloques de K/V en anillo, solapando con cómputoNinguno sobre el nº de cabezastorch.distributed.tensor.experimental.context_parallel; cp_size en accelerate. torchtitan usa una variante que hace all-gather de K/V (KVAllGatherCPFlexInnerAttention)
Ulysses / ALSTall-to-all: cada GPU calcula atención completa sobre un subconjunto de cabezasGrado limitado por el nº de cabezas de atencióntorchtitan (UlyssesCPFlexInnerAttention); DeepSpeed (ALST); accelerate con sp_backend="deepspeed"

En torchtitan, CP son dos cambios en tu función de config: fijar el grado y aplicar ContextParallelTransform, que cambia la atención del modelo (flex por defecto en Llama 3) por su versión CP. Es el patrón de llama3_debugmodel_fsdp2_cp2 en torchtitan_recipes/tests/suites/features.py:

python · latam_recipes/latam.py (continúa)
from torchtitan.config.transform import ContextParallelTransform, apply_transforms
from torchtitan.models.common.attention import FlexInnerAttention
from torchtitan.models.common.attention.cp_attention import KVAllGatherCPFlexInnerAttention


def llama3_8b_latam_cpt_32k() -> Trainer.Config:
    config = llama3_8b_latam_cpt(seq_len=32_768)
    config.parallelism.data_parallel_shard_degree = 2
    config.parallelism.context_parallel_degree = 4      # 2 × 4 = 8 GPUs
    return apply_transforms(config, [
        ContextParallelTransform(
            inner_attention_map={FlexInnerAttention: KVAllGatherCPFlexInnerAttention},
        ),
    ])

CP multiplica con FSDP: 8 GPUs con CP=4 dejan 2 grupos de shard de datos (NGPU tiene que ser el producto de los grados). Para Ulysses, cambia el destino del mapa por UlyssesCPFlexInnerAttention: comunica menos, pero su grado queda limitado por el número de cabezas (Llama 3.1 8B tiene 32 de query y 8 de K/V).

import math

import torch
from datasets import load_dataset
from transformers import AutoModelForCausalLM, AutoTokenizer

SETS = {
    "es": ("HuggingFaceFW/fineweb-2", "spa_Latn", "test"),
    "pt": ("HuggingFaceFW/fineweb-2", "por_Latn", "test"),
    "en": ("HuggingFaceFW/fineweb-edu", "sample-10BT", "train"),  # control de olvido (no es held-out estricto)
}


@torch.no_grad()
def perplexity(model, tok, texts, max_len=2048):
    nll, n = 0.0, 0
    for t in texts:
        ids = tok(t, return_tensors="pt", truncation=True, max_length=max_len).input_ids.to(model.device)
        if ids.shape[1] < 2:
            continue
        loss = model(input_ids=ids, labels=ids).loss
        nll += loss.item() * (ids.shape[1] - 1)
        n += ids.shape[1] - 1
    return math.exp(nll / n)


texts = {k: [ex["text"] for ex in load_dataset(r, name=c, split=s, streaming=True).take(300)]
         for k, (r, c, s) in SETS.items()}
for model_id in ["Qwen/Qwen3-0.6B-Base", "./export-hf"]:
    tok = AutoTokenizer.from_pretrained(model_id)
    model = AutoModelForCausalLM.from_pretrained(model_id, dtype=torch.bfloat16, device_map="cuda").eval()
    print(model_id, {k: round(perplexity(model, tok, v), 2) for k, v in texts.items()})
    del model
    torch.cuda.empty_cache()

Lo esperable tras el continual pretraining: la perplexity baja en es y pt y sube poco en inglés. Si el inglés se dispara, aumenta el replay. Para evaluación downstream en español y portugués, sigue con la guía de evaluación Latam.
