Filtrado de Calidad de Datos

Pipeline completo para filtrar documentos de baja calidad de un corpus web: reglas heurísticas, clasificador de idioma fastText, clasificador de calidad fastText y filtro de perplejidad KenLM.

1. Por qué el filtrado importa

DCLM (2024) demostró experimentalmente que la calidad del pipeline de datos supera a la arquitectura del modelo: un modelo más pequeño entrenado con datos bien filtrados supera a uno más grande con datos crudos. El filtrado es posiblemente el componente con mayor ROI en el pipeline.

Flujo típico: corpus crudo → filtro de idioma → reglas Gopher/C4 → clasificador de calidad → filtro de perplejidad → corpus limpio para training
Guías relacionadas: para un clasificador de calidad multilingüe basado en embeddings (alternativa a los clasificadores fastText solo-inglés) ver JQL: filtrado multilingüe de calidad; para eliminar contenido tóxico del corpus ver Filtrado de toxicidad en pretraining es/pt.

2. Reglas heurísticas (Gopher / C4)

Filtros rápidos basados en estadísticas del texto. Eliminan spam, listas de palabras, contenido autogenerado y páginas rotas.

python — reglas_heuristicas.py
import json


def passes_gopher_rules(text: str) -> bool:
    """
    Reglas inspiradas en el paper de Gopher (DeepMind, 2022).
    Returns True si el documento pasa el filtro.
    """
    words = text.split()
    n_words = len(words)
    n_chars = len(text)

    # 1. Longitud mínima y máxima
    if not (50 <= n_words <= 100_000):
        return False

    # 2. Ratio de caracteres alfabéticos
    alpha_chars = sum(1 for c in text if c.isalpha())
    if alpha_chars / n_chars < 0.70:
        return False

    # 3. Palabras promedio no demasiado largas (spam de SEO)
    avg_word_len = sum(len(w) for w in words) / n_words
    if not (3 <= avg_word_len <= 10):
        return False

    # 4. No más del 10% de líneas terminan con "..."
    lines = text.split("\n")
    ellipsis_lines = sum(1 for l in lines if l.rstrip().endswith("..."))
    if len(lines) > 0 and ellipsis_lines / len(lines) > 0.10:
        return False

    # 5. No más del 30% de líneas son duplicadas (listas repetitivas)
    non_empty_lines = [l for l in lines if l.strip()]
    if len(non_empty_lines) > 5:
        unique_ratio = len(set(non_empty_lines)) / len(non_empty_lines)
        if unique_ratio < 0.70:
            return False

    # 6. Ratio de símbolos (filtrar código o contenido técnico irrelevante)
    symbol_ratio = sum(1 for c in text if c in "{}[]<>|#@") / n_chars
    if symbol_ratio > 0.05:
        return False

    return True


# Aplicar a dataset
def filter_corpus(input_path: str, output_path: str):
    kept, removed = 0, 0
    with open(input_path, encoding="utf-8") as fin, open(output_path, "w", encoding="utf-8") as fout:
        for line in fin:
            doc = json.loads(line)
            if passes_gopher_rules(doc["text"]):
                fout.write(line)
                kept += 1
            else:
                removed += 1
    print(f"Conservados: {kept:,} ({kept/(kept+removed):.1%}) | Eliminados: {removed:,}")

python — deteccion_idioma.py
import fasttext

lid_model = fasttext.load_model("lid.176.bin")

def detect_language(text: str, threshold: float = 0.65) -> tuple[str, float]:
    """Detecta el idioma de un texto. Retorna (lang_code, confidence)."""
    # fastText requiere texto sin saltos de línea
    clean = text.replace("\n", " ").strip()[:500]  # solo primeros 500 chars
    predictions = lid_model.predict(clean, k=1)
    lang = predictions[0][0].replace("__label__", "")
    confidence = predictions[1][0]
    return lang, confidence


def is_spanish_or_portuguese(text: str, threshold: float = 0.65) -> bool:
    lang, conf = detect_language(text)
    return lang in ("es", "pt") and conf >= threshold


# Filtrar corpus por idioma (una sola predicción por documento)
def filter_by_language(docs: list[dict], target_langs=("es", "pt"), threshold=0.65):
    kept = []
    for doc in docs:
        lang, conf = detect_language(doc["text"])
        if lang in target_langs and conf >= threshold:
            kept.append({**doc, "language": lang, "language_score": conf})
    return kept
Alternativa: GlotLID. GlotLID cubre 1.600+ idiomas y distingue mejor variedades cercanas; en datatrove basta con LanguageFilter(languages=["spa_Latn", "por_Latn"], backend="glotlid") (códigos ISO 639-3 + escritura).

4. Clasificador de calidad fastText

Entrenamos un clasificador binario: Wikipedia/textos de calidad como clase positiva vs. web crawl crudo como clase negativa. Permite filtrar documentos que "se parecen a" texto de referencia.

python — entrenar_clasificador_calidad.py
import fasttext
from pathlib import Path

# Preparar datos de entrenamiento en formato fastText
# Positivos: Wikipedia en español/portugués
# Negativos: muestra del web crawl crudo

def prepare_training_file(
    positives: list[str],   # textos de Wikipedia
    negatives: list[str],   # textos de web crawl
    output_path: str,
    max_chars: int = 300,
):
    with open(output_path, "w") as f:
        for text in positives:
            clean = text.replace("\n", " ")[:max_chars]
            f.write(f"__label__quality {clean}\n")
        for text in negatives:
            clean = text.replace("\n", " ")[:max_chars]
            f.write(f"__label__low_quality {clean}\n")


# Entrenar el modelo
def train_quality_classifier(train_file: str, val_file: str) -> fasttext.FastText._FastText:
    model = fasttext.train_supervised(
        input=train_file,
        epoch=3,
        lr=0.1,
        wordNgrams=2,        # bigramas para mejor contexto
        dim=256,
        loss="hs",           # hierarchical softmax (más rápido)
        thread=16,
    )
    result = model.test(val_file)
    print(f"N={result[0]:,}, P@1={result[1]:.3f}, R@1={result[2]:.3f}")
    return model


# Usar el clasificador
def score_document(model, text: str) -> float:
    """Retorna probabilidad de que el documento sea de alta calidad."""
    clean = text.replace("\n", " ")[:300]
    pred = model.predict(clean, k=2)
    labels, probs = pred
    quality_idx = labels.index("__label__quality")
    return probs[quality_idx]


# Filtrar corpus
model = fasttext.load_model("quality_classifier_es.bin")
HIGH_QUALITY_THRESHOLD = 0.5

filtered = [doc for doc in corpus if score_document(model, doc["text"]) >= HIGH_QUALITY_THRESHOLD]
print(f"Conservados: {len(filtered):,} / {len(corpus):,} ({len(filtered)/len(corpus):.1%})")

5. Filtro de perplejidad con KenLM

Documentos con perplejidad muy alta contienen texto incoherente o generado automáticamente. Un modelo de lenguaje de n-gramas entrenado en Wikipedia permite detectarlos eficientemente.

bash — instalar KenLM
pip install kenlm sentencepiece datatrove

# Los modelos KenLM de CCNet (5-gramas, Wikipedia) esperan texto normalizado
# a la CCNet y tokenizado con su SentencePiece: se necesitan ambos archivos.
# datatrove los descarga solos desde el repo HF edugp/kenlm (wikipedia/es, wikipedia/pt).
# Descarga manual equivalente desde CCNet:
wget https://dl.fbaipublicfiles.com/cc_net/lm/es.arpa.bin
wget https://dl.fbaipublicfiles.com/cc_net/lm/es.sp.model
python — filtro_perplejidad.py
import numpy as np
from datatrove.utils.perplexity import KenlmModel

# Normalización CCNet (minúsculas, dígitos→0, sin diacríticos, puntuación unicode)
# + tokenización SentencePiece + score KenLM, igual que en el pipeline original
lm = {
    "es": KenlmModel.from_pretrained("wikipedia", "es"),
    "pt": KenlmModel.from_pretrained("wikipedia", "pt"),
}


def compute_perplexity(text: str, lang: str) -> float:
    """Perplejidad del documento según el KenLM de Wikipedia del idioma dado."""
    if len(text.split()) < 10:
        return float("inf")
    return lm[lang].get_perplexity(text)


# Los umbrales absolutos no son transferibles entre modelos/idiomas:
# calibrarlos por percentiles sobre una muestra del propio corpus
# (CCNet partió cada idioma en terciles head/middle/tail).
def filter_by_perplexity(
    docs: list[dict],
    lang: str,
    low_pct: float = 1,     # quita el 1% más bajo (texto repetitivo/plantillas)
    high_pct: float = 70,   # quita el 30% más alto (incoherente/spam)
) -> list[dict]:
    scored = [{**doc, "perplexity": compute_perplexity(doc["text"], lang)} for doc in docs]
    ppls = np.array([d["perplexity"] for d in scored if np.isfinite(d["perplexity"])])
    lo, hi = np.percentile(ppls, [low_pct, high_pct])
    return [d for d in scored if lo <= d["perplexity"] <= hi]

6. Pipeline completo con Datatrove

python — pipeline_filtrado.py
from datatrove.executor import LocalPipelineExecutor
from datatrove.pipeline.readers import JsonlReader
from datatrove.pipeline.writers import JsonlWriter
from datatrove.pipeline.filters import (
    GopherQualityFilter,
    GopherRepetitionFilter,
    LanguageFilter,
    C4QualityFilter,
)
from datatrove.utils.typeshelper import Languages

# Por defecto Gopher/C4 usan language=english y stop words en inglés
# ("the", "be", "to", ...): sin esto se descarta casi todo el texto es/pt.
# Se corre un pipeline por idioma, con su tokenizador y sus stop words.
CONFIG = {
    "es": (Languages.spanish, ["de", "la", "que", "el", "en", "y", "a", "los"]),
    "pt": (Languages.portuguese, ["de", "a", "o", "que", "e", "do", "da", "em"]),
}

for lid_code, (language, stop_words) in CONFIG.items():
    executor = LocalPipelineExecutor(
        pipeline=[
            JsonlReader(data_folder="data/raw/", glob_pattern="**/*.jsonl"),

            # 1. Filtro de idioma (lid.176: códigos "es" / "pt")
            LanguageFilter(languages=[lid_code], language_threshold=0.65),

            # 2. Reglas Gopher de calidad básica
            GopherQualityFilter(
                min_doc_words=50,
                max_doc_words=100_000,
                min_avg_word_length=3,
                max_avg_word_length=10,
                max_symbol_word_ratio=0.1,
                stop_words=stop_words,
                language=language,
            ),

            # 3. Filtro de repetición (valores de Gopher)
            GopherRepetitionFilter(
                dup_n_grams=((5, 0.15), (6, 0.14), (7, 0.13), (8, 0.12), (9, 0.11), (10, 0.10)),
                top_n_grams=((2, 0.20), (3, 0.18), (4, 0.16)),
                language=language,
            ),

            # 4. Filtros C4 adicionales
            C4QualityFilter(
                filter_no_terminal_punct=True,
                min_num_sentences=3,
                language=language,
            ),

            # Guardar corpus filtrado
            JsonlWriter(output_folder=f"data/filtered/{lid_code}/"),
        ],
        tasks=64,
        workers=16,
        logging_dir=f"logs/filtering/{lid_code}/",
    )
    executor.run()
Orden importa: Poner los filtros más baratos primero (longitud, idioma) elimina documentos antes de aplicar filtros costosos (perplejidad, clasificadores ML). Esto reduce el tiempo total de procesamiento.

7. DCLM: clasificador OH2.5 + ELI5 vs RefinedWeb

La innovación principal de DCLM fue usar datos de instrucción (OpenHermes 2.5 + ELI5 Reddit) como positivos en vez de Wikipedia. La hipótesis: si quieres que el modelo escriba como en instrucciones de alta calidad, entrena el clasificador con esos ejemplos. El umbral usado en DCLM-Baseline es 0.018112: no es permisivo, sino que corresponde a quedarse con aprox. el 10% superior de los documentos según el score (las probabilidades P(hq) están muy sesgadas hacia 0, por eso el valor absoluto es bajo). Lo correcto es fijar el umbral por percentil sobre tu propio corpus.

Ojo: fasttext_oh_eli5 se entrenó solo con texto en inglés. Para es/pt hay que entrenar un clasificador propio con positivos en español/portugués (instrucciones traducidas o curadas) o usar un clasificador multilingüe como JQL o FineWeb-Edu multilingüe.

bash — preparar datos y entrenar clasificador DCLM

cd dclm && pip install -e .

# Negativos: muestra aleatoria de RefinedWeb / CommonCrawl

# training_data.txt: una línea por documento con __label__hq o __label__cc
python baselines/train_fasttext_classifier.py \

      model_filename: fasttext_oh_eli5.bin
      key: fasttext_oh_eli5_vs_rw_v2_prob

      key: fasttext_oh_eli5_vs_rw_v2_prob

python — alternativa directa con fasttext
import fasttext

model = fasttext.load_model("fasttext_oh_eli5.bin")

def dclm_score(text: str) -> float:
    """
    Retorna P(hq) según el clasificador DCLM.
    Threshold del paper: 0.018112 (≈ top-10% del pool de DCLM; no es permisivo).
    """
    clean = text.replace("\n", " ").strip()[:300]
    pred = model.predict(clean, k=2)
    labels, probs = pred
    # Si el label "hq" es primero, su probabilidad es probs[0]
    if labels[0] == "__label__hq":
        return float(probs[0])
    return float(1.0 - probs[0])


# Filtrar corpus: umbral por percentil (DCLM-Baseline conservó ~10% superior)
import numpy as np

scores = np.array([dclm_score(doc["text"]) for doc in corpus])
threshold = np.percentile(scores, 90)
filtered = [doc for doc, sc in zip(corpus, scores) if sc >= threshold]
OH2.5 vs Wikipedia como positivos: Wikipedia captura "texto enciclopédico bien escrito". OpenHermes/ELI5 captura "texto explicativo, instructivo, útil para un asistente". Para un LLM de chat/instrucción, el segundo da mejores resultados downstream según los ablaciones de DCLM.

8. Dolma Toolkit — pipeline de OLMo/AllenAI

Dolma es el toolkit de curación de datos de AllenAI, usado para construir el corpus de OLMo. Está escrito en Rust + Python, es muy rápido y usa deduplicación con Bloom filters en Rust. Los "taggers" son módulos Python que anotan atributos sobre cada documento; el paso de mezcla/filtrado aplica los umbrales.

bash — instalar y aplicar taggers
pip install dolma

# Ver todos los taggers disponibles
dolma list

# Taggers clave disponibles:
# gopher_v1                       → reglas de calidad y repetición de Gopher
# c4_v1 / c4_v2                   → reglas del dataset C4 (puntuación terminal, etc.)
# ft_lang_id_en_doc_v2            → LID fastText, pero solo puntúa inglés (no sirve para es/pt)
# jigsaw_hatespeech_document_v2   → contenido de odio (clasificador Jigsaw, inglés)
# pii_regex_v1 / pii_presidio_v1  → detección de PII

# Aplicar taggers sobre el corpus (anota atributos, no filtra todavía).
# Los documentos deben estar bajo una carpeta "documents/"; los atributos
# se escriben en la carpeta hermana "attributes/<experiment>/".
dolma tag \
  --experiment latam \
  --documents "data/documents/*.jsonl.gz" \
  --taggers gopher_v1 c4_v2 \
  --processes 16
python — filtrar con los atributos de Dolma
import gzip
import json

# dolma tag escribe, por cada archivo de documents/, un archivo paralelo en
# attributes/latam/ con {"id", "attributes": {"latam__gopher_v1__word_count": [[start, end, score]], ...}}
# (clave = experimento__tagger__atributo; cada valor es una lista de spans)
# El LID es/pt se hace antes, con datatrove o GlotLID (Dolma solo trae LID para inglés).

def doc_score(attrs: dict, key: str, default: float) -> float:
    spans = attrs.get(key)
    return spans[0][2] if spans else default


def should_keep(attrs: dict) -> bool:
    if doc_score(attrs, "latam__gopher_v1__word_count", 0) < 50:
        return False
    if doc_score(attrs, "latam__gopher_v1__symbol_to_word_ratio", 1.0) > 0.1:
        return False
    return True


with gzip.open("data/documents/part-0000.jsonl.gz", "rt", encoding="utf-8") as fd, \
     gzip.open("data/attributes/latam/part-0000.jsonl.gz", "rt", encoding="utf-8") as fa:
    filtered = [
        json.loads(d) for d, a in zip(fd, fa)
        if should_keep(json.loads(a)["attributes"])
    ]

# Para corpus grandes, usar "dolma mix" con filtros declarados en un YAML
# en vez de este bucle en Python.
Ojo con gopher_v1 en es/pt: sus atributos de "palabras requeridas" usan stop words en inglés; para español/portugués usa solo los atributos independientes del idioma (conteo de palabras, símbolos, repetición) o los filtros por idioma de datatrove.

9. NeMo Curator — filtrado acelerado con GPU

NeMo Curator de NVIDIA acelera el pipeline de filtrado usando RAPIDS/cuDF en GPU. Desde la serie 25.x/26.x los pipelines se arman con Pipeline + stages y se ejecutan sobre Ray (la API Dask con DocumentDataset/nc.Sequential quedó deprecada). La deduplicación fuzzy es ~16× más rápida que la alternativa en CPU. Útil para corpus de decenas de TB donde Datatrove en CPU es demasiado lento.

bash
# CPU: filtros heurísticos y fastText
uv pip install "nemo-curator[text_cpu]"

# GPU (CUDA 12, RAPIDS/cuDF + dedup): cluster con NVIDIA A100/H100
uv pip install "nemo-curator[text_cuda12]"
python — pipeline NeMo Curator
from nemo_curator.pipeline import Pipeline
from nemo_curator.backends.ray_data import RayDataExecutor
from nemo_curator.stages.text.io.reader import JsonlReader
from nemo_curator.stages.text.io.writer import JsonlWriter
from nemo_curator.stages.text.filters import ScoreFilter
from nemo_curator.stages.text.filters.fasttext import FastTextLangId, FastTextQualityFilter
from nemo_curator.stages.text.filters.heuristic import NonAlphaNumericFilter, WordCountFilter
from nemo_curator.stages.text.filters.heuristic.repetition import RepeatedLinesByCharFilter

LANG = "es"  # repetir con "pt"

pipeline = Pipeline(name=f"filtrado_{LANG}")
pipeline.add_stage(JsonlReader(file_paths="data/raw/", fields=["text", "id"]))

# 1. Idioma (lid.176: conserva docs cuyo idioma es LANG con score >= 0.65)
pipeline.add_stage(ScoreFilter(
    filter_obj=FastTextLangId(model_path="lid.176.bin", min_langid_score=0.65, lang=LANG),
    text_field="text",
    score_field="language",
))

# 2. Longitud mínima
pipeline.add_stage(ScoreFilter(
    filter_obj=WordCountFilter(min_words=50),
    text_field="text",
    score_field="word_count",
))

# 3. Ratio no alfanumérico
pipeline.add_stage(ScoreFilter(
    filter_obj=NonAlphaNumericFilter(max_non_alpha_numeric_to_text_ratio=0.25),
    text_field="text",
    score_field="non_alpha",
))

# 4. Caracteres en líneas repetidas
pipeline.add_stage(ScoreFilter(
    filter_obj=RepeatedLinesByCharFilter(max_repeated_lines_char_ratio=0.70),
    text_field="text",
    score_field="repeated_lines",
))

# 5. Clasificador de calidad fastText (sin umbral fijo: muestreo Pareto
#    sobre P(hq), como en GPT-3; label debe coincidir con tu modelo)
pipeline.add_stage(ScoreFilter(
    filter_obj=FastTextQualityFilter(model_path="quality_classifier_es.bin", label="__label__hq", alpha=3),
    text_field="text",
    score_field="quality_score",
))

pipeline.add_stage(JsonlWriter(path=f"data/filtered/{LANG}/"))

# Ejecutar sobre Ray (local o cluster)
pipeline.run(RayDataExecutor(ignore_head_node=True))
GPU vs CPU: Para deduplicación fuzzy de 1T+ tokens, NeMo Curator en GPU es 10-16× más rápido que datasketch en CPU. Para el filtrado de texto puro, la diferencia es menor (~2-4×). La ventaja real es la deduplicación.
