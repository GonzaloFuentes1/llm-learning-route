Anonimización y PII

Detectar y reemplazar información de identificación personal (PII) en corpus de texto antes de entrenamiento. Cubre Microsoft Presidio con soporte para español/portugués y formatos latinoamericanos.

1. Tipos de PII relevantes para Latam

Los modelos que aprenden de texto con PII pueden "memorizar" y reproducir datos personales. La detección y remoción es un requisito legal (LGPD en Brasil; en Chile, la Ley 19.628 modificada por la Ley 21.719, que crea la Agencia de Protección de Datos Personales y entra en vigencia el 1 de diciembre de 2026) y una buena práctica de seguridad.

text — categorías de PII
PII universal:
  - Nombres de personas (PERSON)
  - Emails (EMAIL_ADDRESS)
  - Números de teléfono (PHONE_NUMBER)
  - Direcciones IP (IP_ADDRESS)
  - Fechas de nacimiento (DATE_TIME + contexto)
  - Números de tarjeta de crédito (CREDIT_CARD)
  - URLs personales

PII específica de Latam:
  - RUT (Chile):        12.345.678-5
  - CUIT/CUIL (Arg):   20-12345678-9
  - CPF (Brasil):       123.456.789-09
  - DNI (España/Arg):   12345678A
  - Cédula (Colombia):  1.234.567.890
  - Pasaportes

python — presidio_pipeline.py
from presidio_analyzer import AnalyzerEngine
from presidio_analyzer.nlp_engine import NlpEngineProvider
from presidio_anonymizer import AnonymizerEngine
from presidio_anonymizer.entities import OperatorConfig

# Configurar motor NLP con spaCy español
configuration = {
    "nlp_engine_name": "spacy",
    "models": [
        {"lang_code": "es", "model_name": "es_core_news_lg"},
        {"lang_code": "pt", "model_name": "pt_core_news_lg"},
    ],
}
provider = NlpEngineProvider(nlp_configuration=configuration)
nlp_engine = provider.create_engine()

analyzer = AnalyzerEngine(nlp_engine=nlp_engine, supported_languages=["es", "pt"])
anonymizer = AnonymizerEngine()

# Operadores: reemplazar con tokens genéricos
operators = {
    "PERSON":        OperatorConfig("replace", {"new_value": "<PERSONA>"}),
    "EMAIL_ADDRESS": OperatorConfig("replace", {"new_value": "<EMAIL>"}),
    "PHONE_NUMBER":  OperatorConfig("replace", {"new_value": "<TELEFONO>"}),
    "IP_ADDRESS":    OperatorConfig("replace", {"new_value": "<IP>"}),
    "CREDIT_CARD":   OperatorConfig("replace", {"new_value": "<TARJETA>"}),
    "URL":           OperatorConfig("replace", {"new_value": "<URL>"}),
    "LOCATION":      OperatorConfig("keep"),  # mantener ciudades (útil para Latam)
}


def anonymize_text(text: str, lang: str = "es") -> str:
    """Anonimiza PII en un texto. Retorna el texto con tokens de reemplazo."""
    results = analyzer.analyze(text=text, language=lang)
    if not results:
        return text
    anonymized = anonymizer.anonymize(
        text=text,
        analyzer_results=results,
        operators=operators,
    )
    return anonymized.text


# Ejemplo
texto = "El Dr. Juan García (juan.garcia@hospital.cl, +56 9 1234 5678) vive en Santiago."
print(anonymize_text(texto))
# Output: "El Dr. <PERSONA> (<EMAIL>, <TELEFONO>) vive en Santiago."

3. Reconocedores custom para formatos latinoamericanos

python — recognizers_latam.py
from presidio_analyzer import AnalyzerEngine, Pattern, PatternRecognizer
from presidio_analyzer.nlp_engine import NlpEngineProvider


class RutRecognizer(PatternRecognizer):
    """RUT chileno: 12.345.678-5 o 12345678-5. Valida el dígito verificador (módulo 11)."""

    def validate_result(self, pattern_text: str):
        clean = "".join(c for c in pattern_text if c.isdigit() or c in "kK").upper()
        body, dv = clean[:-1], clean[-1]
        total, factor = 0, 2
        for d in reversed(body):
            total += int(d) * factor
            factor = 2 if factor == 7 else factor + 1
        r = 11 - total % 11
        expected = {10: "K", 11: "0"}.get(r, str(r))
        return dv == expected


class CpfRecognizer(PatternRecognizer):
    """CPF brasileño: 123.456.789-09. Valida los dos dígitos verificadores."""

    def validate_result(self, pattern_text: str):
        nums = [int(c) for c in pattern_text if c.isdigit()]
        if len(nums) != 11 or len(set(nums)) == 1:
            return False
        for n in (9, 10):
            total = sum(nums[i] * (n + 1 - i) for i in range(n))
            if (total * 10) % 11 % 10 != nums[n]:
                return False
        return True


# RUT chileno: 12.345.678-5 o 12345678-5
rut_pattern = Pattern(name="rut_pattern", regex=r"\b\d{1,2}\.?\d{3}\.?\d{3}-[\dkK]\b", score=0.85)
# CUIT/CUIL argentino: 20-12345678-9
cuit_pattern = Pattern(name="cuit_pattern", regex=r"\b(20|23|24|27|30|33|34)-\d{8}-[0-9]\b", score=0.85)
# CPF brasileño: score bajo porque 11 dígitos sueltos suelen ser teléfonos o códigos;
# el checksum de validate_result y el contexto ("cpf") filtran los falsos positivos
cpf_pattern = Pattern(name="cpf_pattern", regex=r"\b\d{3}\.?\d{3}\.?\d{3}-?\d{2}\b", score=0.5)


def latam_recognizers(lang: str) -> list[PatternRecognizer]:
    """Una instancia por idioma: PatternRecognizer usa supported_language="en" por defecto
    y analyze(language=...) solo ejecuta los recognizers de ese idioma."""
    return [
        RutRecognizer(supported_entity="RUT_CL", patterns=[rut_pattern], supported_language=lang,
                      context=["rut", "run", "cédula", "documento"]),
        PatternRecognizer(supported_entity="CUIT_AR", patterns=[cuit_pattern], supported_language=lang,
                          context=["cuit", "cuil", "afip"]),
        CpfRecognizer(supported_entity="CPF_BR", patterns=[cpf_pattern], supported_language=lang,
                      context=["cpf", "cadastro"]),
    ]


# Agregar reconocedores al analyzer (es + pt)
def build_latam_analyzer() -> AnalyzerEngine:
    provider = NlpEngineProvider(nlp_configuration={
        "nlp_engine_name": "spacy",
        "models": [
            {"lang_code": "es", "model_name": "es_core_news_lg"},
            {"lang_code": "pt", "model_name": "pt_core_news_lg"},
        ],
    })
    engine = provider.create_engine()
    analyzer = AnalyzerEngine(nlp_engine=engine, supported_languages=["es", "pt"])

    # Registrar reconocedores custom (una instancia por idioma)
    for lang in ("es", "pt"):
        for rec in latam_recognizers(lang):
            analyzer.registry.add_recognizer(rec)

    # Verificar que RUT/CUIT/CPF quedaron activos en español
    entities = {e for r in analyzer.get_recognizers(language="es") for e in r.supported_entities}
    assert {"RUT_CL", "CUIT_AR", "CPF_BR"} <= entities, entities
    return analyzer

4. Procesar corpus a escala

python — anonimizar_corpus.py
import json
from multiprocessing import Pool
from pathlib import Path
from presidio_anonymizer import AnonymizerEngine
from presidio_anonymizer.entities import OperatorConfig

from recognizers_latam import build_latam_analyzer  # sección 3

anonymizer = AnonymizerEngine()
analyzer = None  # se inicializa en cada worker

def init_worker():
    global analyzer
    analyzer = build_latam_analyzer()

def process_document(doc: dict) -> dict | None:
    lang = doc.get("lang", "es")
    if lang not in ("es", "pt"):
        return None  # idioma no soportado: no se puede garantizar la anonimización
    try:
        results = analyzer.analyze(text=doc["text"], language=lang)
        if results:
            anon = anonymizer.anonymize(
                text=doc["text"],
                analyzer_results=results,
                operators={
                    "PERSON": OperatorConfig("replace", {"new_value": "<PERSONA>"}),
                    "EMAIL_ADDRESS": OperatorConfig("replace", {"new_value": "<EMAIL>"}),
                    "PHONE_NUMBER": OperatorConfig("replace", {"new_value": "<TELEFONO>"}),
                    "IP_ADDRESS": OperatorConfig("replace", {"new_value": "<IP>"}),
                    "RUT_CL": OperatorConfig("replace", {"new_value": "<ID_FISCAL>"}),
                    "CUIT_AR": OperatorConfig("replace", {"new_value": "<ID_FISCAL>"}),
                    "CPF_BR": OperatorConfig("replace", {"new_value": "<ID_FISCAL>"}),
                },
            )
            return {**doc, "text": anon.text, "pii_detected": len(results)}
        return {**doc, "pii_detected": 0}
    except Exception as e:
        # Fallar cerrado: un documento que no se pudo analizar NO pasa sin anonimizar
        print(f"[cuarentena] doc {doc.get('id', '?')}: {type(e).__name__}: {e}")
        return None


def anonymize_corpus(input_file: str, output_file: str, workers: int = 16):
    with open(input_file, encoding="utf-8") as f:
        docs = [json.loads(line) for line in f]

    with Pool(workers, initializer=init_worker) as pool:
        results = pool.map(process_document, docs)

    processed = [d for d in results if d is not None]
    print(f"Descartados (error o idioma no soportado): {len(docs) - len(processed):,}")

    docs_with_pii = sum(1 for d in processed if d.get("pii_detected", 0) > 0)
    print(f"Documentos con PII: {docs_with_pii:,} / {len(docs):,} ({docs_with_pii/len(docs):.1%})")

    with open(output_file, "w", encoding="utf-8") as f:
        for doc in processed:
            f.write(json.dumps(doc, ensure_ascii=False) + "\n")
Limitación importante: Presidio + spaCy no detecta el 100% del PII. Especialmente nombres propios en contextos inusuales o documentos de baja calidad pueden escapar. Valida siempre muestreando manualmente 200-500 documentos post-procesados y ajusta los thresholds según el caso de uso.

5. eu-pii-safeguard — NER multilingüe para PII

tabularisai/eu-pii-safeguard es XLM-RoBERTa-large fine-tuned para detección de PII en los 26 idiomas oficiales de la UE, incluyendo español y portugués. Detecta 42 tipos de entidades (nombre, email, IBAN, condición médica, contraseña, etc.) con F1 global de 97%. Es más preciso que Presidio+spaCy para nombres propios en contextos ambiguos.

python — detección con pipeline HuggingFace
from transformers import pipeline

# Pipeline de token classification (NER)
pii_detector = pipeline(
    "token-classification",
    model="tabularisai/eu-pii-safeguard",
    aggregation_strategy="simple",   # une subwords en entidades completas
    device=0,                        # GPU 0; -1 para CPU
)

text = "Contacta a Ana García en ana.garcia@empresa.cl o al +56 9 8765 4321. Su RUT es 12.345.678-5."

entities = pii_detector(text)
for e in entities:
    print(f"{e['entity_group']:20s} [{e['score']:.2f}]  →  '{e['word']}'")

python — anonimizar reemplazando entidades detectadas
from transformers import pipeline

pii_detector = pipeline(
    "token-classification",
    model="tabularisai/eu-pii-safeguard",
    aggregation_strategy="simple",
    device=0,
)

# Etiquetas reales del modelo (model.config.id2label) → token de reemplazo
LABELS = {l.split("-", 1)[-1] for l in pii_detector.model.config.id2label.values() if l != "O"}
REPLACE_MAP = {
    "FIRSTNAME":          "<PERSONA>",
    "MIDDLENAME":         "<PERSONA>",
    "LASTNAME":           "<PERSONA>",
    "EMAIL":              "<EMAIL>",
    "PHONE_NUMBER":       "<TELEFONO>",
    "IP_ADDRESS":         "<IP>",
    "IBAN":               "<IBAN>",
    "CREDIT_CARD_NUMBER": "<TARJETA>",
    "NATIONAL_ID":        "<ID_FISCAL>",
    "TAX_ID":             "<ID_FISCAL>",
    "PASSWORD":           "<CONTRASENA>",
    "HEALTH_CONDITION":   "<SALUD>",
}
assert set(REPLACE_MAP) <= LABELS, set(REPLACE_MAP) - LABELS

def anonymize_with_xlmr(text: str, threshold: float = 0.85) -> str:
    """Reemplaza PII detectado por eu-pii-safeguard."""
    entities = pii_detector(text)
    # Ordenar de derecha a izquierda para no alterar los offsets
    entities = sorted(entities, key=lambda e: e["start"], reverse=True)
    for ent in entities:
        if ent["score"] < threshold:
            continue
        label = ent["entity_group"]
        replacement = REPLACE_MAP.get(label, f"<{label}>")
        text = text[: ent["start"]] + replacement + text[ent["end"] :]
    return text


# Combinar con Presidio para cobertura máxima:
# 1. Aplicar eu-pii-safeguard (mejor en nombres propios)
# 2. Aplicar Presidio sobre el resultado (mejor en regex: emails, IPs, RUT)
#    anonymize_text viene de presidio_pipeline.py (sección 2); para RUT/CUIT/CPF,
#    construye ahí el analyzer con build_latam_analyzer() (sección 3).
from presidio_pipeline import anonymize_text

def full_anonymization(text: str, lang: str = "es") -> str:
    text = anonymize_with_xlmr(text)            # NER neuronal
    text = anonymize_text(text, lang=lang)      # Regex + patrones Latam
    return text
python — procesar corpus con batching eficiente
from transformers import pipeline

pii_detector = pipeline(
    "token-classification",
    model="tabularisai/eu-pii-safeguard",
    aggregation_strategy="simple",
    device=0,
    batch_size=32,        # batching en GPU para mayor throughput
    stride=64,            # textos largos: ventanas solapadas sobre el texto completo
)

def anonymize_corpus_xlmr(docs: list[dict]) -> list[dict]:
    texts = [d["text"] for d in docs]
    # El pipeline acepta listas directamente. Sin truncar: con stride, los documentos más
    # largos que model_max_length se procesan por ventanas y los offsets son del texto original
    all_entities = pii_detector(texts)
    result = []
    for doc, entities in zip(docs, all_entities):
        text = doc["text"]
        for ent in sorted(entities, key=lambda e: e["start"], reverse=True):
            if ent["score"] >= 0.85:
                label = ent["entity_group"]
                replacement = REPLACE_MAP.get(label, f"<{label}>")  # REPLACE_MAP del bloque anterior
                text = text[: ent["start"]] + replacement + text[ent["end"] :]
        result.append({**doc, "text": text})
    return result
eu-pii-safeguard vs Presidio+spaCy: El modelo neuronal gana en nombres propios en contexto ("la paciente María López") y entidades financieras complejas (IBAN, números de cuenta). Presidio+spaCy gana en formatos estructurados (regex) como RUT, emails, IPs. Para cobertura máxima, aplica ambos en secuencia.
